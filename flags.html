// Build a comparison card that reads the image's natural dimensions once loaded
compareCard = (code) => {
  const info = html`<div style="font-size: 0.85rem; color: #555; margin-top: 6px;">Loading dimensions...</div>`;
  const img = html`<img src="flags/${code}.png" style="width: 100%; max-height: 240px; object-fit: contain; border: 1px solid #ddd; border-radius: 4px; background: #f8f9fa;">`;
  img.onload = () => {
    const ratio = (img.naturalWidth / img.naturalHeight).toFixed(2);
    info.textContent = `Dimensions: ${img.naturalWidth} × ${img.naturalHeight} px (Aspect ratio ${ratio}:1)`;
  };
  return html`<div style="flex: 1; min-width: 240px; padding: 12px; border: 1px solid #eee; border-radius: 6px; text-align: center;">
    <h5 style="margin-bottom: 10px;">${labelFor(code)}</h5>
    ${img}
    ${info}
  </div>`;
}

html`<div style="display: flex; flex-wrap: wrap; gap: 16px; margin-top: 1rem;">
  ${compareCard(compareA)}
  ${compareCard(compareB)}
</div>`
makeQuestion = (score = 0, total = 0, streak = 0) => {
  const shuffled = [...codes].sort(() => Math.random() - 0.5);
  const options = shuffled.slice(0, 4);
  const answer = options[Math.floor(Math.random() * 4)];
  return { score, total, streak, answer, options, picked: null };
}

mutable quiz = makeQuestion()

// Render the quiz UI and update `mutable quiz` when the user picks an option
{
  const q = quiz;
  const container = html`<div style="max-width: 550px; margin-top: 0.5rem;">
    <div style="display: flex; justify-content: space-between; margin-bottom: 10px; font-weight: 600;">
      <span>Score: ${q.score} / ${q.total}</span>
      <span>🔥 Streak: ${q.streak}</span>
    </div>
    <div style="text-align: center; padding: 12px; background: #f8f9fa; border: 1px solid #ddd; border-radius: 6px;">
      <img src="flags/${q.answer}.png" style="max-height: 220px; max-width: 100%; object-fit: contain; border: 1px solid #ccc;">
    </div>
    <div style="display: grid; grid-template-columns: 1fr 1fr; gap: 10px; margin-top: 12px;">
      ${q.options.map((opt) => {
        let bg = "#fff";
        let border = "#ccc";
        if (q.picked !== null) {
          if (opt === q.answer) { bg = "#d1e7dd"; border = "#198754"; }
          else if (opt === q.picked) { bg = "#f8d7da"; border = "#dc3545"; }
        }
        const btn = html`<button style="padding: 10px; border-radius: 6px; border: 2px solid ${border}; background: ${bg}; cursor: ${q.picked === null ? 'pointer' : 'default'}; font-weight: 500;">
          ${labelFor(opt)}
        </button>`;
        if (q.picked === null) {
          btn.onclick = () => {
            const correct = opt === q.answer;
            mutable quiz = {
              ...q,
              picked: opt,
              score: q.score + (correct ? 1 : 0),
              total: q.total + 1,
              streak: correct ? q.streak + 1 : 0
            };
          };
        }
        return btn;
      })}
    </div>
  </div>`;

  if (q.picked !== null) {
    const nextBtn = html`<button style="margin-top: 12px; padding: 8px 16px; border-radius: 6px; border: none; background: #0d6efd; color: white; cursor: pointer; font-weight: 600;">
      Next Flag ➡️
    </button>`;
    nextBtn.onclick = () => {
      mutable quiz = makeQuestion(q.score, q.total, q.streak);
    };
    container.appendChild(nextBtn);
  }
  return container;
}
